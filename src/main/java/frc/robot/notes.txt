  public void rightAuto() {
    double leftDistance = leftEncoder.getDistance();
    double rightDistance = rightEncoder.getDistance();
    distance = Math.abs((leftDistance + rightDistance) / 2);

    // init variables
    double id = getTargetID();
    Pose3d targetPose = LimelightHelpers.getBotPose3d_TargetSpace("limelight");
    distanceToTowerTag = -targetPose.getZ();
    double inputSpeed = 0.0;
    double indexerSpeed = 0.0;
    double forwardSpeed = 0;
    turnSpeed = 0;
    double tXCorrection;
    double angleCorrection;

    // STEP 0: drive backwards
    if (autoStep == 0) {
      if (distance < 30) {
        forwardSpeed = -0.25;
      } else {
        autoStep++;
        leftEncoder.reset();
        rightEncoder.reset();

        turnTimer.reset();
        turnTimer.start();
      }
    }

    // STEP 1: turn 45 degrees
    else if (autoStep == 1) {
      if (m_gyro.getYaw() < 43 && turnTimer.get() < 3.0) {
        turnSpeed = 0.25;
      } else {
        turnSpeed = 0.0;
        autoStep++;
      }
    }
    //Step 2: moves forward to get to the middle of the shooting
    else if(autoStep == 2) {
      if(distance < 30){
        forwardSpeed = 0.25;
      } else{
        autoStep++;
        leftEncoder.reset();
        rightEncoder.reset();

        turnTimer.reset();
        turnTimer.start();
      }
    }
    // Step 3: align the robot for shooting
    else if(autoStep == 3){
      if(m_gyro.getYaw() < -173 && turnTimer.get() < 4.0){
        turnSpeed = -0.25;
      } else{ 
        autoStep++;
        turnSpeed = 0;
        leftEncoder.reset();
        rightEncoder.reset();
      }
    }

    // STEP 4: shoot preload
    else if (autoStep == 4) {
      if (!shootTimerStarted) {
        shootTimer.reset();
        shootTimer.start();
        shootTimerStarted = true;
      }
      inputSpeed = 0.8;
      indexerSpeed = 0.7;

      if (shootTimer.get() > 3) {
        inputSpeed = 0;
        indexerSpeed = 0;
        shootTimerStarted = false;
        autoStep++;

        turnTimer.reset();
        turnTimer.start();
      }
    }

    // STEP 5: drive forward and moves until it sees the april tag
    else if (autoStep == 5) {
      if(id != 16 || distanceToTowerTag > 1 ){
        forwardSpeed = 0.4;
      } else{
        autoStep++;
        leftEncoder.reset();
        rightEncoder.reset();
        alignTimer.reset();
        alignTimer.start();
      }
      }
    

    // STEP 6: aligns the robot with the climb april tag
    else if (autoStep == 6) {
      if(alignTimer.get() < 4) {
        double tx = LimelightHelpers.getTX("limelight");
        gyroAssistEnabled = false;
        gyroCorrection = 0;
        tXCorrection = alignPID.calculate(tx, 0.0);
        tXCorrection = Math.max(-0.4, Math.min(0.4, tXCorrection));
        double angleError = targetPose.getRotation().getZ();
        angleCorrection = anglePID.calculate(angleError, 0.0);
        angleCorrection = Math.max(-0.4, Math.min(0.4, angleCorrection));

        forwardSpeed = tXCorrection;
        turnSpeed = angleCorrection;
      } else{
        autoStep++;
        climbTimer.reset();
        climbTimer.start();
      }
    }

    //step 7, climb
    else if (autoStep == 7) {
      if (climbTimer.get() < 3) {
        climb.set(0.5);
      } else{
        climb.set(0);
      }
      }
    }
